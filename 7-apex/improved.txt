public class CreateAccount {
    public String name {get; set;}
    public String phoneNumber {get; set;}
    public String selectedIndustry {get; set;}
    public String websiteURL {get; set;}
    public List<SelectOption> industryOptions {get; set;}

    public CreateAccount(ApexPages.StandardController controller) {
        // Initialize Industry picklist dynamically
        industryOptions = new List<SelectOption>();
        industryOptions.add(new SelectOption('', '--None--'));
        Schema.DescribeFieldResult fieldResult = Account.Industry.getDescribe();
        for (Schema.PicklistEntry entry : fieldResult.getPicklistValues()) {
            if (entry.isActive()) {
                industryOptions.add(new SelectOption(entry.getValue(), entry.getLabel()));
            }
        }
    }

    public PageReference createAccount() {
        try {
            // Validate inputs
            if (String.isEmpty(name)) {
                ApexPages.addMessage(new ApexPages.Message(
                    ApexPages.Severity.ERROR, 'Account Name is required.'));
                return null;
            }
            if (!String.isEmpty(websiteURL) && !websiteURL.startsWithIgnoreCase('http://') && !websiteURL.startsWithIgnoreCase('https://')) {
                ApexPages.addMessage(new ApexPages.Message(
                    ApexPages.Severity.ERROR, 'Website must start with http:// or https://.'));
                return null;
            }
            if (!String.isEmpty(phoneNumber) && !Pattern.matches('[0-9]+', phoneNumber)) {
                ApexPages.addMessage(new ApexPages.Message(
                    ApexPages.Severity.ERROR, 'Phone number must contain only digits.'));
                return null;
            }

            // Check for duplicate Account Name
            if ([SELECT Id FROM Account WHERE Name = :name LIMIT 1].size() > 0) {
                ApexPages.addMessage(new ApexPages.Message(
                    ApexPages.Severity.WARNING, 'An Account with this name already exists.'));
                return null;
            }

            // Create Account record
            Account acc = new Account(
                Name = name,
                Phone = phoneNumber,
                Industry = selectedIndustry,
                Website = websiteURL
            );
            insert acc;

            // Redirect to the new Account's detail page
            PageReference pg = new PageReference('/' + acc.Id);
            pg.setRedirect(true);
            return pg;
        } catch (DmlException e) {
            ApexPages.addMessage(new ApexPages.Message(
                ApexPages.Severity.ERROR, 'Error creating account: ' + e.getMessage()));
            return null;
        } catch (Exception e) {
            ApexPages.addMessage(new ApexPages.Message(
                ApexPages.Severity.ERROR, 'An unexpected error occurred: ' + e.getMessage()));
            return null;
        }
    }

    public PageReference cancelAccount() {
        // Redirect to Account list view
        return new PageReference('/' + Schema.SObjectType.Account.getKeyPrefix() + '/o');
    }
}
